using Monstertacle.Multisession.Contracts;
using Monstertacle.Multisession.Storage.Sqlite;

internal static class Program
{
    private static async Task<int> Main(string[] args)
    {
        using CancellationTokenSource deadline = new(TimeSpan.FromSeconds(30));
        RuntimeOptions options = new() { StateDirectory = args[0], Retention = new(true, null) };
        await using NativeLease lease = NativeLease.OpenExisting(args[0]);
        try
        {
            await using ISessionStore store = await SqliteStoreFactory.OpenExistingAsync(lease, Guid.ParseExact(args[1], "D"), options, deadline.Token);
        }
        catch (SessionOperationException failure) when (failure.Failure.Code == ErrorCode.UnsupportedStoreVersion)
        {
            Console.WriteLine("PRIOR-READER-REFUSED UnsupportedStoreVersion at store open");
            return 0;
        }
        Console.WriteLine("PRIOR-READER-ACCEPTED");
        return 1;
    }
}
